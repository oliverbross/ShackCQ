cmake_minimum_required(VERSION 3.20)
project(shackcq_core VERSION 0.1.0 LANGUAGES CXX)

option(SHACKCQ_CORE_BUILD_TESTS "Build standalone shared-core tests" ON)
option(SHACKCQ_ENABLE_ASAN "Enable AddressSanitizer for host tests" OFF)
option(SHACKCQ_ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer for host tests" OFF)

set(SHACKCQ_PORTABLE_SOURCES
    portable/src/adif.cpp
    portable/src/cat_parser.cpp
    portable/src/cty.cpp
    portable/src/dx_analysis.cpp
    portable/src/operator_intel.cpp
    portable/src/panadapter_dsp.cpp
    portable/src/protocol.cpp
    portable/src/spot.cpp
    portable/src/sync_queue.cpp
    portable/src/wsjtx_protocol.cpp
)
file(GLOB SHACKCQ_SGP4_SOURCES CONFIGURE_DEPENDS third_party/sgp4/libsgp4/*.cc)

add_library(shackcq_core STATIC src/core.cpp src/features.cpp src/local_receiver.cpp src/receiver.cpp src/remote.cpp src/satellite.cpp src/tci.cpp src/propagation/p533.cpp ${SHACKCQ_PORTABLE_SOURCES} ${SHACKCQ_SGP4_SOURCES})
target_include_directories(shackcq_core PUBLIC include portable/include PRIVATE third_party/sgp4/libsgp4)
target_compile_features(shackcq_core PUBLIC cxx_std_17)
set_target_properties(shackcq_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(shackcq_core PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(shackcq_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(SHACKCQ_CORE_BUILD_TESTS)
    enable_testing()
    add_executable(shackcq_core_tests test/core_tests.cpp)
    target_link_libraries(shackcq_core_tests PRIVATE shackcq_core)
    target_include_directories(shackcq_core_tests PRIVATE third_party/sgp4/libsgp4)
    if(MSVC)
        target_compile_options(shackcq_core_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(shackcq_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME shackcq_core_tests COMMAND shackcq_core_tests)

    add_executable(shackcq_propagation_adapter_tests test/propagation_adapter_tests.cpp)
    target_link_libraries(shackcq_propagation_adapter_tests PRIVATE shackcq_core)
    if(MSVC)
        target_compile_options(shackcq_propagation_adapter_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(shackcq_propagation_adapter_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME shackcq_propagation_adapter_tests COMMAND shackcq_propagation_adapter_tests)

    add_executable(shackcq_lifecycle_stress_tests test/lifecycle_stress_tests.cpp)
    target_link_libraries(shackcq_lifecycle_stress_tests PRIVATE shackcq_core)
    if(MSVC)
        target_compile_options(shackcq_lifecycle_stress_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(shackcq_lifecycle_stress_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME shackcq_lifecycle_stress_tests COMMAND shackcq_lifecycle_stress_tests)

    add_executable(shackcq_receiver_contract_tests test/receiver_contract_tests.cpp)
    target_link_libraries(shackcq_receiver_contract_tests PRIVATE shackcq_core)
    if(MSVC)
        target_compile_options(shackcq_receiver_contract_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(shackcq_receiver_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME shackcq_receiver_contract_tests COMMAND shackcq_receiver_contract_tests)

    add_executable(shackcq_tci_protocol_tests test/tci_protocol_tests.cpp)
    target_link_libraries(shackcq_tci_protocol_tests PRIVATE shackcq_core)
    if(MSVC)
        target_compile_options(shackcq_tci_protocol_tests PRIVATE /W4 /WX /permissive-)

        target_compile_options(shackcq_tci_protocol_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME shackcq_tci_protocol_tests COMMAND shackcq_tci_protocol_tests)

    add_executable(shackcq_local_receiver_tests test/local_receiver_tests.cpp)
    target_link_libraries(shackcq_local_receiver_tests PRIVATE shackcq_core)
    if(MSVC)
        target_compile_options(shackcq_local_receiver_tests PRIVATE /W4 /WX /permissive-)

    target_link_libraries(shackcq_remote_scale_tests PRIVATE shackcq_core)

        target_compile_options(shackcq_remote_scale_tests PRIVATE /W4 /WX /permissive-)

        target_compile_options(shackcq_remote_scale_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)

    set(SHACKCQ_SANITIZER_FLAGS)
    if(SHACKCQ_ENABLE_ASAN)
        list(APPEND SHACKCQ_SANITIZER_FLAGS -fsanitize=address)
    endif()
    if(SHACKCQ_ENABLE_UBSAN)
        list(APPEND SHACKCQ_SANITIZER_FLAGS -fsanitize=undefined)
    endif()
    if(SHACKCQ_SANITIZER_FLAGS)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "ShackCQ host sanitizers require Clang or GCC")
        endif()
        foreach(target shackcq_core shackcq_core_tests shackcq_propagation_adapter_tests shackcq_lifecycle_stress_tests shackcq_receiver_contract_tests shackcq_tci_protocol_tests shackcq_local_receiver_tests shackcq_remote_protocol_tests shackcq_remote_scale_tests)
            target_compile_options(${target} PRIVATE ${SHACKCQ_SANITIZER_FLAGS} -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE ${SHACKCQ_SANITIZER_FLAGS})
        endforeach()
    endif()
